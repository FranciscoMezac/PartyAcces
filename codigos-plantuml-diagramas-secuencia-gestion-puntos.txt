@startuml
title GP-001 - Acumulación de Puntos (Controller → Service → Model ↔ Repository → DB)

actor "Trabajador" as T
participant "dashboard_html" as V
participant "HTTP Router" as R
participant "PuntosController" as C
participant "PuntosService" as S
participant "CuentaPuntos (Model)" as MCta
participant "Movimiento (Model)" as MMov
participant "CuentaRepository" as RC
participant "MovimientoRepository" as RM
database "DBPool" as DB

T -> V: Ingresar {rut, monto}
V -> R: POST /api/puntos/acumular {rut,monto}
R -> C: acumular(req,res)

== Validaciones mínimas en Controller ==
C -> C: validar rut y monto>0
alt inválido
  C --> V: 400 {success:false, message:"Datos inválidos"}
  return
end

== Servicio orquesta ==
C -> S: acumularPuntos(rut, monto)

activate S
' 1) Cargar cuenta como Model con su repo
S -> MCta: new CuentaPuntos({rut}, RC)
MCta --> S: instancia

' 2) El modelo se carga a sí mismo desde el repo
S -> MCta: cargarPorRut()
MCta -> RC: findByRut(rut)
RC -> DB: SELECT * FROM cuentas WHERE rut=$1
DB --> RC: row|null
RC --> MCta: row|null
alt no existe
  MCta --> S: Error("NOT_FOUND")
  S --> C: Error("NOT_FOUND")
  C --> V: 404 {success:false, message:"Cuenta no existe"}
  deactivate S
  return
else existe
  MCta --> S: ok
end

' 3) Reglas de negocio dentro del Modelo
S -> MCta: calcularPuntosAcreditacion(monto)
MCta --> S: puntos
S -> MCta: acreditar(puntos)
note right of MCta
  Actualiza saldo interno y
  persiste usando el Repository
end note
MCta -> RC: actualizarSaldo(rut, nuevoSaldo)
RC -> DB: UPDATE cuentas SET saldo=$1 WHERE rut=$2
DB --> RC: OK
RC --> MCta: OK
MCta --> S: nuevoSaldo

' 4) Registrar movimiento desde el propio Modelo de Movimiento
S -> MMov: new Movimiento({tipo:"COMPRA", rut, monto, puntos}, RM)
MMov --> S: instancia
S -> MMov: registrar()
MMov -> RM: insert(movimiento)
RM -> DB: INSERT INTO movimientos(...)
DB --> RM: OK
RM --> MMov: OK
MMov --> S: OK

S --> C: {success:true, nuevoSaldo, puntos}
deactivate S
C --> V: 201 {success:true, nuevoSaldo, puntos}
@enduml



@startuml
title GP-002 - Canje de Puntos (Controller → Service → Model ↔ Repository → DB)

actor "Trabajador" as T
participant "dashboard_html" as V
participant "HTTP Router" as R
participant "PuntosController" as C
participant "PuntosService" as S
participant "CuentaPuntos (Model)" as MCta
participant "Producto (Model)" as MProd
participant "Movimiento (Model)" as MMov
participant "CuentaRepository" as RC
participant "ProductoRepository" as RP
participant "MovimientoRepository" as RM
database "DBPool" as DB

T -> V: Selecciona canje {rut, productoId}
V -> R: POST /api/puntos/canjear {rut, productoId}
R -> C: canjear(req,res)

== Validaciones en Controller ==
C -> C: validar rut y productoId
alt inválido
  C --> V: 400 {success:false, message:"Datos inválidos"}
  return
end

== Service ==
C -> S: canjear(rut, productoId)
activate S

' 1) Cargar cuenta (Model con repo) y traer estado
S -> MCta: new CuentaPuntos({rut}, RC)
MCta --> S: instancia
S -> MCta: cargarPorRut()
MCta -> RC: findByRut(rut)
RC -> DB: SELECT * FROM cuentas WHERE rut=$1
DB --> RC: row
RC --> MCta: row
MCta --> S: ok

' 2) Cargar producto (Model con repo)
S -> MProd: new Producto({id:productoId}, RP)
MProd --> S: instancia
S -> MProd: cargarPorId()
MProd -> RP: findById(productoId)
RP -> DB: SELECT * FROM productos WHERE id=$1
DB --> RP: row
RP --> MProd: row
MProd --> S: ok

' 3) Reglas de negocio en los Model
S -> MProd: costoEnPuntos()
MProd --> S: costo

S -> MCta: puedeDebitar(costo)?
MCta --> S: ok|no

alt saldo insuficiente
  S --> C: Error("INSUFFICIENT_POINTS")
  C --> V: 409 {success:false, message:"Saldo insuficiente"}
  deactivate S
  return
end

S -> MCta: debitar(costo)
MCta -> RC: actualizarSaldo(rut, nuevoSaldo)
RC -> DB: UPDATE cuentas SET saldo=$1 WHERE rut=$2
DB --> RC: OK
RC --> MCta: OK
MCta --> S: nuevoSaldo

' 4) Registrar movimiento desde el Modelo
S -> MMov: new Movimiento({tipo:"CANJE", rut, productoId, puntos:costo}, RM)
MMov --> S: instancia
S -> MMov: registrar()
MMov -> RM: insert(movimiento)
RM -> DB: INSERT INTO movimientos(...)
DB --> RM: OK
RM --> MMov: OK
MMov --> S: OK

S --> C: {success:true, nuevoSaldo, canje:{productoId, costo}}
deactivate S
C --> V: 201 {success:true, nuevoSaldo, canje:{productoId,costo}}
@enduml


@startuml
title GP-003 - Historial de Transacciones (Controller → Service → Model ↔ Repository → DB)

actor "Usuario" as U
participant "dashboard_html" as V
participant "HTTP Router" as R
participant "PuntosController" as C
participant "PuntosService" as S
participant "Movimiento (Model)" as MMov
participant "MovimientoRepository" as RM
database "DBPool" as DB

U -> V: Ver historial (page, filtros)
V -> R: GET /api/puntos/historial?page=1&...
R -> C: historial(req,res)

C -> C: obtener rut de sesión y validar paginación
C -> S: obtenerHistorial(rut, page, filtros)

activate S
' El Model coordina la lectura a través del repo
S -> MMov: new Movimiento({}, RM)
MMov --> S: instancia
S -> MMov: listarPorRut(rut, page, filtros)
MMov -> RM: findByRutPaginated(rut, page, filtros)
RM -> DB: SELECT fecha,tipo,monto,puntos\nFROM movimientos WHERE rut=$1 ...\nORDER BY fecha DESC LIMIT ... OFFSET ...
DB --> RM: rows
RM --> MMov: {itemsDTO, pageInfo}
MMov --> S: {itemsDTO, pageInfo}

S --> C: {movimientos:itemsDTO, pageInfo}
deactivate S
C --> V: 200 {movimientos, pageInfo}
@enduml


@startuml
title GP-004 - Saldo en Tiempo Real (Controller → Service → Model ↔ Repository → DB + Notificador)

actor "Usuario" as U
participant "dashboard_html" as V
participant "HTTP Router" as R
participant "PuntosController" as C
participant "PuntosService" as S
participant "CuentaPuntos (Model)" as MCta
participant "CuentaRepository" as RC
participant "NotificadorSaldo (SSE/WebSocket)" as N
database "DBPool" as DB

== Consulta inicial ==
U -> V: Abrir panel
V -> R: GET /api/puntos/saldo
R -> C: saldo(req,res)
C -> C: obtener rut de sesión
C -> S: obtenerSaldo(rut)

activate S
S -> MCta: new CuentaPuntos({rut}, RC)
MCta --> S: instancia
S -> MCta: cargarSaldo()
MCta -> RC: getSaldo(rut)
RC -> DB: SELECT saldo FROM cuentas WHERE rut=$1
DB --> RC: saldo
RC --> MCta: saldo
MCta --> S: saldo
S --> C: {saldo}
deactivate S
C --> V: 200 {saldo}

== Notificación tras cambios de saldo (desde GP-001/GP-002) ==
S -> N: publishSaldoActualizado(rut, nuevoSaldo)
N -> V: evento SSE/WebSocket {nuevoSaldo}
@enduml
